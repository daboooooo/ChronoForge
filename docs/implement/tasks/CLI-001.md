class Settings(BaseSettings):

    meta_dir: Path = Path("./meta")
    log_level: str = "INFO"
    fred_api_key: SecretStr | None = None
    sec_contact_email: EmailStr | None = None

    retry_max: int = 5
    quality_block_on: list[str] = ["Q-SCHEMA-001", "Q-PROV-001"]
    normalize_error_threshold: float = 0.10
    rate_overrides_json: str = "{}"

    model_config = SettingsConfigDict(
        env_file=".env",
        env_file_encoding="utf-8",
        case_sensitive=False,
        extra="ignore",
    )

    @property
    def sec_user_agent(self) -> str:
        """派生字段：D08 §1"""
        email = self.sec_contact_email or "research@example.com"
        version = getattr(sys.modules.get("chronoforge"), "__version__", "0.0.0")
        return f"ChronoForge/{version} research ({email})"

    @classmethod
    def load(cls) -> "Settings":
        """单入口加载（架构 02）"""
        return cls()

### 脱敏（logging.py）

```python
import structlog
import json

REDACT_KEYS = frozenset({
    "authorization", "api_key", "apikey", "token",
    "cookie", "set-cookie", "x-api-key", "fred_api_key",
})

def redact(obj: Any) -> Any:
    """
    递归 dict/list 脱敏（D08 §4）。
    命中 REDACT_KEYS → "***"；URL query 参数同规则掩码。
    """
    if isinstance(obj, dict):
        return {
            k: "***" if k.lower() in REDACT_KEYS else redact(v)
            for k, v in obj.items()
        }
    elif isinstance(obj, list):
        return [redact(item) for item in obj]
    elif isinstance(obj, str):
        # URL query 参数脱敏
        if "?" in obj:
            base, query = obj.split("?", 1)
            redacted_query = "&".join(
                f"{k}=***" if k.lower() in REDACT_KEYS else p
                for p in query.split("&")
                for k in [p.split("=", 1)[0]]
            )
            return f"{base}?{redacted_query}"
        return obj
    elif isinstance(obj, SecretStr):
        return "***"
    else:
        return obj

def redact_processor(logger, method_name, event_dict):
    """structlog processor 链末位"""
    if "event" in event_dict:
        event_dict["event"] = redact(event_dict["event"])
    if "context" in event_dict:
        event_dict["context"] = redact(event_dict["context"])
    if "exc_info" in event_dict and event_dict["exc_info"]:
        if isinstance(event_dict["exc_info"], BaseException):
            event_dict["exc_info"] = redact(format_exception(event_dict["exc_info"]))
    return event_dict


def setup_logging(level: str = "INFO") -> None:
    """配置 structlog（D08 §3）"""
    structlog.configure(
        processors=[
            structlog.contextvars.merge_contextvars,
            structlog.processors.add_log_level,
            structlog.processors.StackInfoRenderer(),
            structlog.dev.set_exc_info,
            redact_processor,
            structlog.processors.TimeStamper(fmt="iso"),
            structlog.processors.JSONRenderer(),
        ],
        wrapper_class=structlog.make_filtering_bound_logger(
            getattr(logging, level.upper(), logging.INFO)
        ),
        context_class=dict,
        logger_factory=structlog.PrintLoggerFactory(),
        cache_logger_on_first_use=True,
    )
```
